1. สร้างเงินสำรองฉุกเฉินให้ครบก่อน (Day 7) · 2. เลือกสัดส่วนที่ผ่านคำถาม "ฉากวิกฤตแล้วจะขายไหม" · 3. หากองดัชนีต้นทุนต่ำสำหรับแต่ละช่อง เช็ก TER รวมชั้น master fund และ tracking difference · 4. ตั้งซื้ออัตโนมัติรายเดือนทันทีที่เงินเดือนเข้า · 5. คัด IPS ในกล่องข้างบนไปเก็บไว้ ทบทวนปีละครั้ง และแก้เฉพาะเมื่อชีวิตเปลี่ยน · ถ้าจะใช้สิทธิลดหย่อน (RMF/SSF/ThaiESG, Day 13) ให้เลือกกองที่ใส่ลงในช่องเดิมของพอร์ต ไม่ใช่เพิ่มช่องใหม่เพราะอยากได้ส่วนลดภาษี

กล่องโค้ด (ข้ามได้ ไม่นับในเวลาอ่าน) 
import numpy as np
E   = np.array([.07, .08, .03, .03])          # หุ้นไทย หุ้นโลก พันธบัตร ทอง
S   = np.array([.18, .16, .05, .15])
TER = np.array([.005, .005, .004, .006])
R   = np.array([[1, .6, .1, 0], [.6, 1, .1, .1], [.1, .1, 1, .2], [0, .1, .2, 1]])
C   = np.outer(S, S) * R

w = np.array([.25, .45, .25, .05])
e = w @ ((1 + E) * (1 - TER) - 1)
s = np.sqrt(w @ C @ w)
real = (1 + e) / 1.027 - 1
m, v = (1 + real) ** (1/12), 0.0
for _ in range(360):
    v = (v + 3000) * m
print(f"E {e:.2%}  σ {s:.2%}  ปีแย่ {e-2*s:.2%}  30 ปี {v:,.0f} บาท (บาทวันนี้)")
